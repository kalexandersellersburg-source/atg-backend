import React, { useState, useEffect } from 'react';
import { Clock, Tv, ChevronLeft, ChevronRight, RefreshCw, Wifi, CheckCircle, AlertCircle, Settings } from 'lucide-react';

export default function TVGuide() {
  const [currentTime, setCurrentTime] = useState(new Date());
  const [selectedChannel, setSelectedChannel] = useState(null);
  const [currentDay, setCurrentDay] = useState(0);
  const [epgData, setEpgData] = useState({});
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const [liveDataChannels, setLiveDataChannels] = useState(new Set());
  const [viewMode, setViewMode] = useState('now');
  const [serverUrl, setServerUrl] = useState('');
  const [showSettings, setShowSettings] = useState(false);

  useEffect(() => {
    const timer = setInterval(() => setCurrentTime(new Date()), 60000);
    return () => clearInterval(timer);
  }, []);

  // Channel mappings with Schedules Direct station IDs
  const channels = [
    { id: 'fox-news', name: 'Fox News Channel', logo: '📰', category: 'News', stationId: '16374' },
    { id: 'amc', name: 'AMC', logo: '🎬', category: 'Entertainment', philo: true, stationId: '10021' },
    { id: 'ae', name: 'A&E', logo: '📺', category: 'Entertainment', philo: true, stationId: '10093' },
    { id: 'hgtv', name: 'HGTV', logo: '🏠', category: 'Lifestyle', philo: true, stationId: '11097' },
    { id: 'food', name: 'Food Network', logo: '🍳', category: 'Lifestyle', philo: true, stationId: '11021' },
    { id: 'comedy', name: 'Comedy Central', logo: '😂', category: 'Entertainment', philo: true, stationId: '11867' },
    { id: 'mtv', name: 'MTV', logo: '🎵', category: 'Entertainment', philo: true, stationId: '10986' },
    { id: 'discovery', name: 'Discovery Channel', logo: '🌍', category: 'Documentary', philo: true, stationId: '11150' },
  ];

  const fetchSchedulesDirectData = async () => {
    if (!serverUrl) {
      setError('Please set your server URL in settings');
      setShowSettings(true);
      return;
    }

    setLoading(true);
    setError(null);
    const liveChannels = new Set();

    try {
      // Generate date range (today + 4 days)
      const dates = [];
      for (let i = 0; i < 5; i++) {
        const date = new Date();
        date.setDate(date.getDate() + i);
        dates.push(date.toISOString().split('T')[0]);
      }

      // Get station IDs
      const stationIds = channels.map(ch => ch.stationId);

      console.log('Fetching schedules from proxy server...');
      
      // Fetch schedules via proxy
      const scheduleResponse = await fetch(`${serverUrl}/api/schedules`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ stationIds, dates })
      });

      if (!scheduleResponse.ok) {
        const errorData = await scheduleResponse.json().catch(() => ({}));
        throw new Error(errorData.error || `Server error: ${scheduleResponse.status}`);
      }

      const schedules = await scheduleResponse.json();

      if (!schedules || schedules.length === 0) {
        throw new Error('No schedule data received from server');
      }

      // Parse schedules and collect program IDs
      const allData = {};
      const allProgramIds = new Set();

      schedules.forEach(schedule => {
        if (!schedule.programs || schedule.programs.length === 0) return;

        const stationId = schedule.stationID;
        if (!allData[stationId]) {
          allData[stationId] = [];
        }

        schedule.programs.forEach(prog => {
          allProgramIds.add(prog.programID);
          
          const startTime = new Date(prog.airDateTime);
          const stopTime = new Date(startTime.getTime() + prog.duration * 1000);

          allData[stationId].push({
            programID: prog.programID,
            title: prog.programID,
            desc: '',
            start: startTime,
            stop: stopTime,
            new: prog.new || false,
            md5: prog.md5
          });
        });

        liveChannels.add(stationId);
      });

      // Fetch program details in batches
      if (allProgramIds.size > 0) {
        console.log(`Fetching ${allProgramIds.size} program details...`);
        
        const programIds = Array.from(allProgramIds);
        const batchSize = 500;
        
        for (let i = 0; i < programIds.length; i += batchSize) {
          const batch = programIds.slice(i, i + batchSize);
          
          const programResponse = await fetch(`${serverUrl}/api/programs`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ programIds: batch })
          });

          if (programResponse.ok) {
            const programs = await programResponse.json();
            
            // Update program titles and descriptions
            Object.keys(allData).forEach(stationId => {
              allData[stationId].forEach(prog => {
                const details = programs.find(p => p.programID === prog.programID);
                if (details) {
                  prog.title = details.titles?.[0]?.title120 || prog.programID;
                  prog.desc = details.descriptions?.description100?.[0]?.description || 
                             details.descriptions?.description1000?.[0]?.description || '';
                }
              });
            });
          }
        }
      }

      setEpgData(allData);
      setLiveDataChannels(liveChannels);
      setError(null);
      setLoading(false);

      console.log('✅ Schedules Direct data loaded successfully!');
    } catch (err) {
      console.error('Fetch error:', err);
      setError(`Error: ${err.message}. Using demo data.`);
      setLoading(false);
      loadEnhancedDemoData();
    }
  };

  const loadEnhancedDemoData = () => {
    const now = new Date();
    const baseDate = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    
    const createDailySchedule = (dayOffset, startHour, programs) => {
      let currentTime = new Date(baseDate);
      currentTime.setDate(baseDate.getDate() + dayOffset);
      currentTime.setHours(startHour, 0, 0, 0);
      
      return programs.map(prog => {
        const start = new Date(currentTime);
        const stop = new Date(currentTime.getTime() + prog.duration * 60000);
        currentTime = new Date(stop);
        return { ...prog, start, stop };
      });
    };
    
    const create5DaySchedule = (startHour, programs) => {
      let allSchedules = [];
      for (let day = 0; day < 5; day++) {
        allSchedules = allSchedules.concat(createDailySchedule(day, startHour, programs));
      }
      return allSchedules;
    };

    const foxNewsPrograms = [
      { title: 'Fox & Friends First', duration: 60, desc: 'Early morning news' },
      { title: 'Fox & Friends', duration: 180, desc: 'Morning news show' },
      { title: 'America\'s Newsroom', duration: 120, desc: 'Morning news coverage' },
      { title: 'The Faulkner Focus', duration: 60, desc: 'Midday news' },
      { title: 'Outnumbered', duration: 60, desc: 'Panel discussion' },
      { title: 'America Reports', duration: 120, desc: 'Afternoon news' },
      { title: 'The Story', duration: 60, desc: 'Evening news' },
      { title: 'Your World', duration: 60, desc: 'Business news' },
      { title: 'The Five', duration: 60, desc: 'Panel show' },
      { title: 'Special Report', duration: 60, desc: 'Political news' },
      { title: 'The Ingraham Angle', duration: 60, desc: 'Commentary' },
      { title: 'Jesse Watters Primetime', duration: 60, desc: 'Analysis' },
      { title: 'Hannity', duration: 60, desc: 'Commentary' },
      { title: 'Gutfeld!', duration: 60, desc: 'Late-night show' },
      { title: 'Fox News @ Night', duration: 60, desc: 'Late news' },
    ];

    const genericPrograms = [
      { title: 'Early Morning', duration: 120, desc: 'Morning block' },
      { title: 'Mid Morning', duration: 180, desc: 'Morning shows' },
      { title: 'Afternoon', duration: 240, desc: 'Afternoon programming' },
      { title: 'Prime Time', duration: 180, desc: 'Evening shows' },
      { title: 'Late Night', duration: 120, desc: 'Late programming' },
    ];

    const demo = {
      '16374': create5DaySchedule(5, foxNewsPrograms),
      '10021': create5DaySchedule(6, genericPrograms),
      '10093': create5DaySchedule(7, genericPrograms),
      '11097': create5DaySchedule(6, genericPrograms),
      '11021': create5DaySchedule(7, genericPrograms),
      '11867': create5DaySchedule(8, genericPrograms),
      '10986': create5DaySchedule(8, genericPrograms),
      '11150': create5DaySchedule(6, genericPrograms),
    };
    
    setEpgData(demo);
  };

  const getChannelSchedule = (channelId, stationId) => {
    const programs = epgData[stationId] || [];
    const dayOffset = currentDay * 24 * 60 * 60 * 1000;
    const targetDate = new Date(Date.now() + dayOffset);
    
    const dayPrograms = programs.filter(prog => {
      const progDate = new Date(prog.start);
      return progDate.getDate() === targetDate.getDate() &&
             progDate.getMonth() === targetDate.getMonth() &&
             progDate.getFullYear() === targetDate.getFullYear();
    }).sort((a, b) => a.start - b.start);

    if (viewMode === 'now' && currentDay === 0) {
      const now = currentTime;
      const twoHoursFromNow = new Date(now.getTime() + 2 * 60 * 60 * 1000);
      
      return dayPrograms.filter(prog => {
        return prog.stop && prog.stop >= now && prog.start <= twoHoursFromNow;
      });
    }

    return dayPrograms;
  };

  const isShowLive = (startTime, stopTime) => {
    if (currentDay !== 0) return false;
    const now = currentTime.getTime();
    return now >= startTime.getTime() && (!stopTime || now < stopTime.getTime());
  };

  const formatTime = (date) => {
    return date.toLocaleTimeString('en-US', { 
      hour: 'numeric', 
      minute: '2-digit',
      hour12: true 
    });
  };

  const getDuration = (start, stop) => {
    if (!stop) return '60';
    return Math.round((stop - start) / 60000);
  };

  useEffect(() => {
    loadEnhancedDemoData();
  }, []);

  const days = ['Today', 'Tomorrow', 'Day 3', 'Day 4', 'Day 5'];

  return (
    <div className="min-h-screen bg-gray-900 text-white pb-20">
      <div className="bg-gradient-to-r from-blue-600 to-blue-800 p-4 shadow-lg sticky top-0 z-10">
        <div className="max-w-7xl mx-auto">
          <div className="flex items-center justify-between gap-2">
            <div className="flex items-center gap-2">
              <Tv className="w-7 h-7" />
              <div>
                <h1 className="text-xl font-bold">TV Guide</h1>
                <p className="text-xs text-blue-200">Schedules Direct • Live EPG</p>
              </div>
            </div>
            
            <div className="flex items-center gap-2">
              <button
                onClick={() => setShowSettings(!showSettings)}
                className="p-2 bg-blue-700 rounded hover:bg-blue-600 transition"
              >
                <Settings className="w-4 h-4" />
              </button>
              <div className="flex items-center gap-1 bg-blue-700 px-2 py-1 rounded text-sm">
                <Clock className="w-4 h-4" />
                <span className="font-semibold">
                  {currentTime.toLocaleTimeString('en-US', { 
                    hour: 'numeric', 
                    minute: '2-digit' 
                  })}
                </span>
              </div>
            </div>
          </div>

          {showSettings && (
            <div className="mt-3 bg-blue-900 rounded-lg p-3">
              <h3 className="font-bold mb-2 flex items-center gap-2">
                <Settings className="w-4 h-4" />
                Server Settings
              </h3>
              <input
                type="text"
                placeholder="https://your-server.railway.app"
                value={serverUrl}
                onChange={(e) => setServerUrl(e.target.value)}
                className="w-full px-3 py-2 bg-gray-800 text-white rounded border border-gray-600 focus:border-blue-500 outline-none text-sm"
              />
              <p className="text-xs text-blue-200 mt-2">
                Enter your deployed server URL (from Railway/Render)
              </p>
            </div>
          )}

          {error && (
            <div className="mt-3 bg-yellow-600 bg-opacity-20 border border-yellow-500 rounded-lg p-2 flex items-start gap-2 text-sm">
              <AlertCircle className="w-4 h-4 mt-0.5 flex-shrink-0" />
              <span className="text-xs">{error}</span>
            </div>
          )}

          {liveDataChannels.size > 0 && !error && (
            <div className="mt-2 bg-green-600 bg-opacity-20 border border-green-500 rounded-lg p-2 flex items-center gap-2">
              <CheckCircle className="w-4 h-4 flex-shrink-0" />
              <span className="text-xs">✓ Schedules Direct: {liveDataChannels.size} channels loaded</span>
            </div>
          )}
          
          <div className="flex items-center gap-1 mt-3 overflow-x-auto">
            <button 
              onClick={() => setCurrentDay(Math.max(0, currentDay - 1))}
              disabled={currentDay === 0}
              className="p-1.5 bg-blue-700 rounded disabled:opacity-50 flex-shrink-0"
            >
              <ChevronLeft className="w-4 h-4" />
            </button>
            {days.map((day, idx) => (
              <button
                key={idx}
                onClick={() => setCurrentDay(idx)}
                className={`px-3 py-1.5 rounded-lg transition flex-shrink-0 text-sm ${
                  currentDay === idx 
                    ? 'bg-white text-blue-600 font-bold' 
                    : 'bg-blue-700'
                }`}
              >
                {day}
              </button>
            ))}
            <button 
              onClick={() => setCurrentDay(Math.min(4, currentDay + 1))}
              disabled={currentDay === 4}
              className="p-1.5 bg-blue-700 rounded disabled:opacity-50 flex-shrink-0"
            >
              <ChevronRight className="w-4 h-4" />
            </button>
          </div>

          {currentDay === 0 && (
            <div className="flex gap-2 mt-2">
              <button
                onClick={() => setViewMode('now')}
                className={`flex-1 px-3 py-2 rounded-lg transition text-sm ${
                  viewMode === 'now'
                    ? 'bg-white text-blue-600 font-bold'
                    : 'bg-blue-700'
                }`}
              >
                Now + Next 2 Hours
              </button>
              <button
                onClick={() => setViewMode('full')}
                className={`flex-1 px-3 py-2 rounded-lg transition text-sm ${
                  viewMode === 'full'
                    ? 'bg-white text-blue-600 font-bold'
                    : 'bg-blue-700'
                }`}
              >
                Full Day
              </button>
            </div>
          )}

          <button
            onClick={fetchSchedulesDirectData}
            disabled={loading || !serverUrl}
            className="w-full mt-2 flex items-center justify-center gap-2 bg-blue-700 hover:bg-blue-600 px-3 py-2 rounded-lg transition disabled:opacity-50 text-sm"
          >
            <RefreshCw className={`w-4 h-4 ${loading ? 'animate-spin' : ''}`} />
            {loading ? 'Loading from SD...' : serverUrl ? 'Refresh Schedule' : 'Set Server URL'}
          </button>
        </div>
      </div>

      <div className="max-w-7xl mx-auto p-3">
        {loading && Object.keys(epgData).length === 0 ? (
          <div className="text-center py-12">
            <RefreshCw className="w-12 h-12 animate-spin mx-auto mb-4 text-blue-500" />
            <p className="text-lg">Loading from Schedules Direct...</p>
            <p className="text-sm text-gray-400 mt-2">Via proxy server</p>
          </div>
        ) : (
          <div className="space-y-3">
            {channels.map(channel => {
              const schedule = getChannelSchedule(channel.id, channel.stationId);
              const hasData = schedule.length > 0;
              const hasLiveData = liveDataChannels.has(channel.stationId);
              
              return (
                <div 
                  key={channel.id} 
                  className="bg-gray-800 rounded-lg overflow-hidden shadow-lg"
                >
                  <div className="bg-gradient-to-r from-gray-700 to-gray-800 p-3 flex items-center justify-between">
                    <div className="flex items-center gap-2">
                      <span className="text-2xl">{channel.logo}</span>
                      <div>
                        <h2 className="text-base font-bold">{channel.name}</h2>
                        <div className="flex gap-1 mt-0.5 flex-wrap">
                          <span className="text-xs bg-blue-600 px-1.5 py-0.5 rounded">
                            {channel.category}
                          </span>
                          {channel.philo && (
                            <span className="text-xs bg-purple-600 px-1.5 py-0.5 rounded">
                              Philo
                            </span>
                          )}
                          {!channel.philo && (
                            <span className="text-xs bg-red-600 px-1.5 py-0.5 rounded">
                              Fox News
                            </span>
                          )}
                          {hasLiveData && (
                            <span className="text-xs bg-green-600 px-1.5 py-0.5 rounded flex items-center gap-1">
                              <CheckCircle className="w-3 h-3" /> SD Live
                            </span>
                          )}
                        </div>
                      </div>
                    </div>
                    <button
                      onClick={() => setSelectedChannel(
                        selectedChannel === channel.id ? null : channel.id
                      )}
                      className="px-3 py-1.5 bg-blue-600 hover:bg-blue-700 rounded-lg transition text-sm"
                    >
                      {selectedChannel === channel.id ? 'Hide' : 'View'}
                    </button>
                  </div>

                  <div className={`transition-all ${selectedChannel === channel.id ? 'max-h-96 overflow-y-auto' : 'max-h-40'}`}>
                    {!hasData ? (
                      <div className="p-3 text-center text-gray-400 text-sm">
                        <p>No schedule available</p>
                      </div>
                    ) : (
                      schedule.map((program, idx) => {
                        const isLive = isShowLive(program.start, program.stop);
                        return (
                          <div 
                            key={idx}
                            className={`p-3 border-b border-gray-700 ${
                              isLive ? 'bg-red-900 bg-opacity-30' : ''
                            }`}
                          >
                            <div className="flex justify-between items-start gap-2">
                              <div className="flex-1 min-w-0">
                                <div className="flex items-center gap-2 mb-1">
                                  <span className="text-xs text-gray-400 font-mono">
                                    {formatTime(program.start)}
                                  </span>
                                  {isLive && (
                                    <span className="bg-red-600 text-white text-xs px-1.5 py-0.5 rounded animate-pulse">
                                      LIVE
                                    </span>
                                  )}
                                  {program.new && (
                                    <span className="bg-blue-600 text-white text-xs px-1.5 py-0.5 rounded">
                                      NEW
                                    </span>
                                  )}
                                </div>
                                <h3 className="text-sm font-semibold">{program.title}</h3>
                                {program.desc && (
                                  <p className="text-xs text-gray-400 mt-1 line-clamp-2">{program.desc}</p>
                                )}
                                <p className="text-xs text-gray-500 mt-0.5">{getDuration(program.start, program.stop)} min</p>
                              </div>
                            </div>
                          </div>
                        );
                      })
                    )}
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </div>
    </div>
  );
}
